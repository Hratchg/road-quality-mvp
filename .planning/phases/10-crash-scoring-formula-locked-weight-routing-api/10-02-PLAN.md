<!-- This is the EXACT SQL to put in CRASH_UPDATE_SQL. From RESEARCH.md §Code Examples → Example 3, with the column-name fix applied (length_m / 1000.0, not length_km). -->
<!-- DO NOT inline severity weights as numeric literals — pass them as %(name)s parameters so they stay in sync with backend/app/scoring.py constants. -->

```sql
WITH per_seg AS (
    SELECT
        rs.id AS segment_id,
        rs.length_m,
        SUM(
            CASE c.severity
                WHEN 'fatal' THEN %(fatal_w)s
                WHEN 'injury' THEN %(injury_w)s
                WHEN 'pdo' THEN %(pdo_w)s
                ELSE 0
            END
        ) AS raw_sum
    FROM road_segments rs
    LEFT JOIN crash_records c ON c.snapped_segment_id = rs.id
    GROUP BY rs.id, rs.length_m
),
per_seg_capped AS (
    SELECT
        segment_id,
        length_m,
        LEAST(raw_sum, %(fatal_cap)s) AS capped_sum,
        LEAST(raw_sum, %(fatal_cap)s) /
            GREATEST(length_m / 1000.0, 0.05) AS raw_per_km
    FROM per_seg
),
p95v AS (
    SELECT PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY raw_per_km) AS p95
    FROM per_seg_capped
    WHERE raw_per_km > 0   -- Pitfall 6: p95 over crash-bearing segments only
)
UPDATE segment_scores ss
SET crash_norm = LEAST(
        1.0,
        psc.raw_per_km / NULLIF((SELECT p95 FROM p95v), 0)
    ),
    updated_at = NOW()
FROM per_seg_capped psc
WHERE ss.segment_id = psc.segment_id
  AND psc.raw_per_km > 0;   -- D-10-09: zero-crash segments stay at DEFAULT 0.0
```

<!-- 2. Length is `rs.length_m / 1000.0` (km) — column is METERS per Migration 001 + Pitfall 5 + CORRECTED CONTEXT D-10-07. -->

<!-- 5. NULLIF((SELECT p95 FROM p95v), 0) defends against the all-zero case (no crashes anywhere). When NULLIF returns NULL, the LEAST(...) is NULL, and the UPDATE's outer WHERE filter (raw_per_km > 0) means we never touch those rows anyway — but defensive coding mandates the NULLIF. -->
